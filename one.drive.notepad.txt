# 1. DEFINE VARIABLES
$ManagerFQDN = "YOUR_MANAGER_FQDN"
$ManagerPort = 443

Write-Host "--- MISSION VERIFICATION: APP VOLUMES CONNECTIVITY ---" -ForegroundColor Yellow

# 2. NETWORK PERSISTENCE CHECK (TCP 443)
Write-Host "[1/4] Checking Network Line of Sight to Manager..." -ForegroundColor Cyan
$TestNet = Test-NetConnection -ComputerName $ManagerFQDN -Port $ManagerPort
if ($TestNet.TcpTestSucceeded) {
    Write-Host "[✓] SUCCESS: Port $ManagerPort is open on $ManagerFQDN." -ForegroundColor Green
} else {
    Write-Host "[X] FAILED: Cannot reach Manager on $ManagerPort. Check Firewall/NSGs." -ForegroundColor Red
}

# 3. SERVICE STATUS CHECK
Write-Host "`n[2/4] Checking App Volumes Services..." -ForegroundColor Cyan
$SvService = Get-Service -Name "svservice" -ErrorAction SilentlyContinue
if ($SvService.Status -eq "Running") {
    Write-Host "[✓] SUCCESS: 'svservice' is Running." -ForegroundColor Green
} else {
    Write-Host "[X] FAILED: 'svservice' is NOT Running." -ForegroundColor Red
}

# 4. REGISTRY CONFIGURATION CHECK
Write-Host "`n[3/4] Checking Registry Parameters..." -ForegroundColor Cyan
$RegPath = "HKLM:\System\CurrentControlSet\Services\svservice\Parameters"
$CurrentMgr = (Get-ItemProperty -Path $RegPath -ErrorAction SilentlyContinue).ManagerAddress

if ($CurrentMgr -eq $ManagerFQDN) {
    Write-Host "[✓] SUCCESS: Registry ManagerAddress matches ($CurrentMgr)." -ForegroundColor Green
} else {
    Write-Host "[X] WARNING: Registry shows '$CurrentMgr' instead of '$ManagerFQDN'." -ForegroundColor Yellow
}

# 5. DRIVER STATUS CHECK (Filter Driver)
Write-Host "`n[4/4] Checking App Volumes Filter Driver (svdriver)..." -ForegroundColor Cyan
$SvDriver = fltmc instances | Select-String "svdriver"
if ($SvDriver) {
    Write-Host "[✓] SUCCESS: 'svdriver' is attached and active." -ForegroundColor Green
} else {
    Write-Host "[X] FAILED: 'svdriver' is NOT detected in the filter stack. Reboot may be required." -ForegroundColor Red
}
